### 2.2 上下文装配核心（最值钱的一段代码）

```python
@dataclass
class ContextChunk:
    id: str
    content: str
    kind: str                  # code / history / tool_result / spec
    tokens: int
    embedding: list[float] | None
    meta: dict
    age: int = 0               # 距离当前的步数
    is_hard: bool = False      # 硬约束（永不压缩/丢弃）

def score(
    c: ContextChunk,
    task_emb: list[float],
    current_step: str,
    selected: list[ContextChunk],
    weights: Weights,
) -> float:
    relevance = cosine(c.embedding, task_emb) if c.embedding else 0.0
    recency   = math.exp(-weights.lambda_ * c.age)
    dependency = 1.0 if current_step in c.meta.get("deps", []) else 0.0
    density    = c.meta.get("info_units", 1) / max(c.tokens, 1)
    redundancy = max(
        (cosine(c.embedding, s.embedding) for s in selected if s.embedding),
        default=0.0,
    )
    return (
        weights.w1 * relevance
        + weights.w2 * recency
        + weights.w3 * dependency
        + weights.w4 * density
        - weights.w5 * redundancy
    )

def assemble(candidates: list[ContextChunk], budget: TokenBudget) -> list[ContextChunk]:
    # 硬约束优先
    hard = [c for c in candidates if c.is_hard]
    selected, used = list(hard), sum(c.tokens for c in hard)
    pool = [c for c in candidates if not c.is_hard]

    while pool and used < budget.total:
        best = max(pool, key=lambda c: score(c, task_emb, step, selected, W))
        if used + best.tokens > budget.total:
            pool.remove(best)
            continue
        selected.append(best); used += best.tokens; pool.remove(best)

    return placement_order(selected)   # 硬约束→头，最新状态→尾

def placement_order(chunks):
    hard = [c for c in chunks if c.is_hard]
    recent = sorted([c for c in chunks if not c.is_hard], key=lambda c: -c.age)[:2]
    middle = [c for c in chunks if c not in hard and c not in recent]
    return hard + middle + recent      # 对抗 lost-in-the-middle
```

### 2.4 Verifier 的关键设计

```python
class VerifierAgent(BaseAgent):
    async def run(self, ctx: AgentContext) -> AgentMessage:
        # ★ 关键：上下文里不包含 Coder 的自我解释
        # 只包含：验收标准 + 实际产出 + 客观测试结果
        checks = []
        for criterion in ctx.acceptance_criteria:
            result = await self.verify_one(criterion, ctx.artifact, ctx.test_result)
            checks.append(result)

        passed = all(c.passed for c in checks)
        return AgentMessage(
            type=MessageType.FEEDBACK,
            payload=FeedbackPayload(
                verdict="PASS" if passed else "REJECT",
                failed_criteria=[c.text for c in checks if not c.passed],
                evidence={...},
                suggestions=await self.suggest_fixes(checks),
                lesson=await self.reflect(checks) if not passed else None,
            ),
        )
```

## 5. 评测执行脚本骨架

```python
async def run_eval(dataset: list[EvalCase], strategy: str) -> EvalReport:
    results = []
    for case in dataset:
        task = await create_task(case.goal)
        trace = await orchestrate(task, strategy=strategy)  # 可切换策略
        score = await evaluate(
            trace=trace,
            expected=case.expected,
            metrics=["success", "first_pass", "cost", "trajectory"],
        )
        results.append(score)
    return EvalReport.aggregate(results)

# 对照实验
# strategy="naive"    → 全量上下文 + 单一模型
# strategy="compress" → 仅压缩
```

## 6. 起步顺序建议

1. **先跑通最小闭环**：单个 Coder Agent + 沙箱执行 + 简单任务
2. **加编排**：Orchestrator + DAG + 2-3 个 Agent
3. **做上下文工程**：这是核心，值得花最多时间
4. **加可靠性**：Verifier + Reflexion + Checkpoint
5. **加评测**：没有评测就无法量化，尽早做（哪怕简陋）
6. **前端可视化**：上下文可视化优先，演示价值最高

> **重要提醒**：不要一开始就追求全功能。先跑通闭环、尽早引入评测，
> 否则后面无法判断改动是变好还是变坏。
